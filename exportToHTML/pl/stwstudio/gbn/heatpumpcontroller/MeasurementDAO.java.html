package pl.stwstudio.gbn.heatpumpcontroller;

import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.stereotype.Repository;

import java.time.LocalDate;
import java.util.List;
import java.util.Map;


@Repository
public class MeasurementDAO {
    private JdbcTemplate jdbcTemplate;

    @Autowired
    public MeasurementDAO(JdbcTemplate jdbcTemplate){
        this.jdbcTemplate = jdbcTemplate;
    }

    public void add (Measurement measurement){
        String sql = "INSERT INTO measurement VALUES(?,?,?,?,?,?,?)";
        jdbcTemplate.update(sql,
                measurement.getIdMeasurement(),
                measurement.getLocalDateTime(),
                measurement.getPower(),
                measurement.getTemp(0),
                measurement.getTemp(1),
                measurement.getTemp(2),
                measurement.getTemp(3));
    }

    public List <Map<String, Object>> showLastMeasurement (LocalDate localDate){
        String sql = "SELECT * FROM measurement WHERE DATE >= ? ' 00:00:00' AND DATE <= ? ' 23:59:59'" ;
        return jdbcTemplate.queryForList(sql,new Object [] {localDate.toString(),localDate.toString()});
    }
}
