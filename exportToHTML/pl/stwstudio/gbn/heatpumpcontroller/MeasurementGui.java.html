package pl.stwstudio.gbn.heatpumpcontroller;

import com.vaadin.flow.component.Text;
import com.vaadin.flow.component.button.Button;
import com.vaadin.flow.component.datepicker.DatePicker;
import com.vaadin.flow.component.datetimepicker.DateTimePicker;
import com.vaadin.flow.component.orderedlayout.VerticalLayout;
import com.vaadin.flow.component.textfield.TextArea;
import com.vaadin.flow.component.textfield.TextField;
import com.vaadin.flow.router.Route;
import org.springframework.beans.factory.annotation.Autowired;

import java.time.Duration;
import java.util.List;
import java.util.Map;

@Route
public class MeasurementGui extends VerticalLayout {
    TextField textField1 ;
    TextField textField2 ;
    TextField textField3 ;
    Text text;
    Button button1;
    Button button2;
    TextArea textArea;
    TextArea textArea2;
    private MeasurementDAO measurementDAO;

@Autowired
    public MeasurementGui(MeasurementDAO measurementDAO){
        this.textField1 = new TextField("ID");
        this.textField2 = new TextField("DATA");
        this.textField3 = new TextField("Power");
        this.text = new Text("Dodawanie danych do bazy");
        this.button1 = new Button("Dodaj zapis");
        this.button2 = new Button("Pokaz wynik wg daty");
        this.textArea = new TextArea();
        this.textArea2 = new TextArea();
        this.measurementDAO = measurementDAO;

        DateTimePicker dateTimePicker = new DateTimePicker();
        dateTimePicker.setLabel("Wybierz date pomiaru");
        dateTimePicker.setStep((Duration.ofMinutes(1)));

        DatePicker.DatePickerI18n singleFormatI18n = new DatePicker.DatePickerI18n();
        singleFormatI18n.setDateFormat("yyyy-MM-dd");
        DatePicker singleFormatDatePicker = new DatePicker("Select a date:");
        singleFormatDatePicker.setI18n(singleFormatI18n);

        button1.addClickListener(x->{
                    Measurement measurement = new Measurement();
                    measurement.setIdMeasurement(Integer.parseInt(textField1.getValue()));
                    measurement.setLocalDateTime(dateTimePicker.getValue());
                    measurement.setPower(Double.parseDouble(textField3.getValue()));
                    measurementDAO.add(measurement);
                });
        button2.addClickListener(x->{
            List<Map<String,Object>> map = measurementDAO.showLastMeasurement(singleFormatDatePicker.getValue());
            textArea.setValue(singleFormatDatePicker.getValue().toString());
            textArea2.setValue(map.toString());
        });

        Charts charts = new Charts();
        HorizontalBar horizontalBar = new HorizontalBar();
        AreaChartExample areaChartExample = new AreaChartExample();

        add(text, textField1,dateTimePicker, singleFormatDatePicker, textField3,button1, button2, textArea, textArea2, charts.build(), horizontalBar.build(), areaChartExample.build());


    }

}
